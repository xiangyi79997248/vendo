const SHEET_NAME = "工作表1"; // 若你的工作表名稱不同請自行修改 function doGet(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({error: "找不到工作表: " + SHEET_NAME})) .setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); if (data.length <= 1) { return ContentService.createTextOutput(JSON.stringify({ vendors: [], engineering: [] })) .setMimeType(ContentService.MimeType.JSON); } // A2 儲存格存放的是整包 JSON 字串 var jsonString = data[1][0]; if (!jsonString) { return ContentService.createTextOutput(JSON.stringify({ vendors: [], engineering: [] })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(jsonString).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { var lock = LockService.getScriptLock(); // 嘗試取得鎖定，最多等待 15 秒，避免多人或多裝置同時寫入衝突 var isLocked = lock.tryLock(15000); if (!isLocked) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "伺服器忙線中，請稍後再試"})) .setMimeType(ContentService.MimeType.JSON); } try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (!sheet) { throw new Error("找不到工作表: " + SHEET_NAME); } var postData = e.postData.contents; // 驗證是否為合法的 JSON 格式，避免寫入毀損資料 JSON.parse(postData); sheet.clear(); sheet.appendRow(["JSON_Data"]); // A1 標題 sheet.appendRow([postData]); // A2 存放整包系統資料 SpreadsheetApp.flush(); // 強制寫入試算表 return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
